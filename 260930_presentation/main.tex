\vspace{3mm}
\begin{block}{Rule of thumb}
Weights take about 1 GB per billion parameters at 8 bit and half of that at 4 bit, before any room for context. A 744B model at 8 bit is roughly 744 GB, which is why it sits alone on the 1.5 TB node.
\end{block}

\begin{frame}{A coding agent is a model plus a harness}{The model decides what to do next, and the harness is the part that actually does it}
\begin{columns}[T,onlytextwidth]
\begin{column}{.47\textwidth}
  \begin{block}{Model: decides}
  \begin{itemize}\itemsep4pt
    \item Reads the task and the code
    \item Chooses the next edit or tool call
    \item Makes sense of what comes back
  \end{itemize}
  \end{block}
  \centering\footnotesize GPT, Claude, Qwen, DeepSeek, GLM
\end{column}
\hfill
\begin{column}{.47\textwidth}
  \begin{exampleblock}{Harness: acts}
  \begin{itemize}\itemsep4pt
    \item Gathers context, offers tools
    \item Edits files and runs the tests
    \item Keeps state and permissions
  \end{itemize}
  \end{exampleblock}
  \centering\footnotesize Codex, Claude Code, OpenCode, Mistral Vibe
\end{column}
\end{columns}
\vspace{4mm}
\centering
{\small You ask $\rightarrow$ harness builds context $\rightarrow$ model picks an action $\rightarrow$ harness runs it $\rightarrow$ repeat}\\[2mm]
\textbf{The model is the engine. The harness is the car built around it.}
\src{https://openai.com/index/unrolling-the-codex-agent-loop/}{OpenAI, \emph{Unrolling the Codex agent loop}}
\end{frame}

\begin{frame}{The model has learned its harness}{Post-training runs inside the agent loop, and Codex is the clearest example}
\small
\begin{center}
\begin{tikzpicture}[>=stealth, every node/.style={font=\small}, node distance=1.4cm]
  \node[draw=ACINred,fill=ACINred!6,rounded corners,minimum width=2.2cm,minimum height=.9cm] (task) {Coding task};
  \node[draw=TUblue,fill=TUblue!6,rounded corners,minimum width=2.5cm,minimum height=.9cm,right=of task] (agent) {Model + tools};
  \node[draw=ACINred,fill=ACINred!6,rounded corners,minimum width=2.6cm,minimum height=.9cm,right=of agent] (test) {Test result};
  \draw[->,thick] (task) -- (agent);
  \draw[->,thick] (agent) -- (test);
  \draw[->,thick] (test.south) .. controls +(0,-.7) and +(0,-.7) .. node[below]{try again until the tests pass} (agent.south);
\end{tikzpicture}
\end{center}
\vspace{1mm}
\begin{itemize}\itemsep5pt
  \item Coding models no longer learn only from prompt and answer. They learn from whole trajectories: inspect, edit, run the tests, read the failure, try again.
  \item OpenAI says this openly. codex-1 and GPT-5-Codex were trained with reinforcement learning on real coding tasks, and GPT-5-Codex is tuned for Codex itself.
  \item You can see it in the scores. On Terminal-Bench 4.0, Grok 4.7 gets about 26\% in a neutral harness and 38\% in xAI's own. Opus 5.5 gets 59.6\% and 66.4\%.
\end{itemize}
\srcmulti{\href{https://openai.com/index/gpt-5-system-card-addendum-gpt-5-codex/}{OpenAI, GPT-5-Codex system-card addendum}, \href{https://proceedings.mlr.press/v267/pan25g.html}{SWE-Gym (ICML 2025)}, \href{https://artificialanalysis.ai/evaluations/terminalbench-4-0}{Terminal-Bench 4.0}}
\end{frame}

\begin{frame}{Why routers exist}{One API in front of many models and many providers}
\begin{center}
\begin{tikzpicture}[>=stealth, every node/.style={font=\small}]
  \node[draw=ACINgray,rounded corners,minimum width=2cm,minimum height=.9cm] (app) at (0,0) {Your harness};
  \node[draw=ACINred,fill=ACINred!6,rounded corners,minimum width=2cm,minimum height=.9cm] (router) at (4,0) {Router};
  \node[draw=TUblue,rounded corners,minimum width=3.4cm] (p1) at (8.5,1.1) {Provider A: cheapest};
  \node[draw=TUblue,rounded corners,minimum width=3.4cm] (p2) at (8.5,0) {Provider B: fastest};
  \node[draw=TUblue,rounded corners,minimum width=3.4cm] (p3) at (8.5,-1.1) {Provider C: in the EU};
  \draw[->,thick] (app) -- node[above]{one API} (router);
  \draw[->] (router) -- (p1.west); \draw[->] (router) -- (p2.west); \draw[->] (router) -- (p3.west);
\end{tikzpicture}
\end{center}
\vspace{2mm}
\begin{itemize}\itemsep4pt
  \item The \textbf{model} is which network answers. The \textbf{provider} is who runs it, and the same open-weight model is often sold by several at different prices, speeds and regions.
  \item A router gives you one integration and one bill. It picks a provider for each request and switches to another when one is down.
  \item So the full stack has five layers: harness, router, model, provider, region.
\end{itemize}
\srcmulti{\href{https://openrouter.ai/docs/quickstart}{OpenRouter quickstart}, \href{https://www.eurouter.ai/docs/concepts/routing}{EUrouter routing docs}}
\end{frame}

\begin{frame}{What we would do}{Three things to take away}
\begin{enumerate}\itemsep9pt
  \item \textbf{Start with what is already here.} Get an AQUEDUCT key, point OpenCode or Pi at it, and try DeepSeek V4 Flash for coding. Use Qwen 3.6 when you need image input or many calls in parallel.
  \item \textbf{Judge the pair, never the model alone.} A score belongs to a model, a harness and a reasoning budget together. Compare within one benchmark version and one setup.
  \item \textbf{Match the bill to the job.} For interactive work on frontier models, a \$20 subscription comes first. For automation or your own harness, use an API, and check provider, region and retention before you send unpublished code.
\end{enumerate}
